def get_random_genre():

    # 此处简化为一个硬编码的流派列表，实际应用中可动态抓取

def get_artist_by_genre(genre):
    url = "https://musicbrainz.org/ws/2/artist"
    params = {
        "query": f'tag:{genre}',
        "fmt": "json",
        "limit": 100
    }

        if artists:

            chosen = random.choice(artists)
